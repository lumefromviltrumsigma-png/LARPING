--========== DRAGGING ==========-
local dragging, dragInput, dragStart, startPos
Topbar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)
Topbar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

--========== OPEN / CLOSE ==========-
local open = true
local isAnimating = false

local function ToggleUI()
    if isAnimating then return end
    isAnimating = true

    if open then
        local t = TweenService:Create(Main, TweenInfo.new(0.28, Enum.EasingStyle.Quint, Enum.EasingDirection.In), {
            Size = UDim2.new(0, 210, 0, 0),
        })
        t:Play()
        t.Completed:Connect(function()
            Main.Visible = false
            open = false
            isAnimating = false
        end)
    else
        Main.Visible = true
        Main.Size = UDim2.new(0, 210, 0, 0)
        local t = TweenService:Create(Main, TweenInfo.new(0.34, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, 210, 0, 150),
        })
        t:Play()
        t.Completed:Connect(function()
            open = true
            isAnimating = false
        end)
    end
end

-- 1. ESP
local ESP_Enabled = false

local function RemoveESP(player)
    if player.Character then
        local head = player.Character:FindFirstChild("Head")
        if head and head:FindFirstChild("LarpESP_Billboard") then head.LarpESP_Billboard:Destroy() end
        if player.Character:FindFirstChild("LarpESP_Highlight") then player.Character.LarpESP_Highlight:Destroy() end
    end
end

local function ApplyESP(player)
    if not ESP_Enabled or player == LocalPlayer then return end
    local function CharacterAdded(char)
        if not char or not ESP_Enabled then return end
        local humanoid = char:WaitForChild("Humanoid", 5)
        local head = char:WaitForChild("Head", 5)
        if not humanoid or not head then return end
        if not head:FindFirstChild("LarpESP_Billboard") then
            local billboard = Instance.new("BillboardGui")
            billboard.Name = "LarpESP_Billboard"
            billboard.Adornee = head
            billboard.Size = UDim2.new(0, 150, 0, 40)
            billboard.StudsOffset = Vector3.new(0, 2.5, 0)
            billboard.AlwaysOnTop = true
            billboard.Parent = head
            local nameLabel = Instance.new("TextLabel", billboard)
            nameLabel.Size = UDim2.new(1, 0, 0, 18)
            nameLabel.BackgroundTransparency = 1
            nameLabel.Text = player.Name
            nameLabel.TextColor3 = CyanColor
            nameLabel.TextSize = 12
            nameLabel.Font = Enum.Font.GothamBold
            nameLabel.TextStrokeTransparency = 0.2
            local hpLabel = Instance.new("TextLabel", billboard)
            hpLabel.Name = "HPLabel"
            hpLabel.Size = UDim2.new(1, 0, 0, 16)
            hpLabel.Position = UDim2.new(0, 0, 0, 18)
            hpLabel.BackgroundTransparency = 1
            hpLabel.Text = "HP: " .. math.floor(humanoid.Health) .. " / " .. math.floor(humanoid.MaxHealth)
            hpLabel.TextColor3 = CyanColor
            hpLabel.TextSize = 10
            hpLabel.Font = Enum.Font.GothamMedium
            hpLabel.TextStrokeTransparency = 0.2
            humanoid.HealthChanged:Connect(function(health)
                hpLabel.Text = "HP: " .. math.floor(health) .. " / " .. math.floor(humanoid.MaxHealth)
            end)
        end
        if not char:FindFirstChild("LarpESP_Highlight") then
            local highlight = Instance.new("Highlight")
            highlight.Name = "LarpESP_Highlight"
            highlight.Adornee = char
            highlight.FillColor = CyanColor
            highlight.FillTransparency = 0.5
            highlight.OutlineColor = CyanColor
            highlight.OutlineTransparency = 0
            highlight.Parent = char
        end
    end
    if player.Character then task.spawn(function() CharacterAdded(player.Character) end) end
    player.CharacterAdded:Connect(CharacterAdded)
end

local harmfulEffectsList = {
    "RagdollStun","Stun","Stunned","StunEffect","StunHit",
    "Knockback","Knockdown","Knockout","Dazed","Paralyzed",
    "Freeze","Frozen","Sleep","Slow","Slowed","Root","Rooted"
}
local RemoveStunEnabled = false
local HeartbeatConn = nil

local function clearHarmfulEffects()
    local char = LocalPlayer.Character
    if not char then return end
    for _, n in ipairs(harmfulEffectsList) do
        local e = char:FindFirstChild(n)
        if e then pcall(function() e:Destroy() end) end
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and hum:GetAttribute("Stunned") then hum:SetAttribute("Stunned", false) end
end

local function toggle_remove_stun(enabled)
    RemoveStunEnabled = enabled and true or false
    if HeartbeatConn then HeartbeatConn:Disconnect() HeartbeatConn = nil end
    if RemoveStunEnabled then
        clearHarmfulEffects()
        HeartbeatConn = RunService.Heartbeat:Connect(function()
            if RemoveStunEnabled then clearHarmfulEffects() end
        end)
    end
    PlayToggleSound()
end

-- 3. Boxing Machine Farm
local AutoPunchEnabled = false
local _AutoPunchOriginal = nil
local AutoStartLoop = nil
local CachedPrompt = nil
local LastFireTime = 0
local FIRE_COOLDOWN = 3.5

local function find_punch_prompt()
    if CachedPrompt and CachedPrompt.Parent and CachedPrompt.Enabled then return CachedPrompt end
    CachedPrompt = nil
    local roots = {
        Workspace:FindFirstChild("Map"), Workspace:FindFirstChild("Interactive"),
        Workspace:FindFirstChild("Objects"), Workspace:FindFirstChild("Machines"),
        Workspace:FindFirstChild("Arcade"), Workspace,
    }
    for _, root in ipairs(roots) do
        if not root then continue end
        local p = root:FindFirstChild("PunchPrompt", true) or root:FindFirstChild("StrengthPrompt", true) or root:FindFirstChild("TestPrompt", true)
        if p and p:IsA("ProximityPrompt") then CachedPrompt = p return p end
    end
    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("ProximityPrompt") then
            local n = string.lower(d.Name)
            local pn = d.Parent and string.lower(d.Parent.Name) or ""
            if n:find("punch") or n:find("strength") or n:find("test")
                or pn:find("punch") or pn:find("strength") or pn:find("tester") then
                CachedPrompt = d return d
            end
        end
    end
    return nil
end

local function fire_prompt(prompt)
    if not prompt or not prompt.Parent then return false end
    if fireproximityprompt then return pcall(fireproximityprompt, prompt) end
    local saved = prompt.HoldDuration
    prompt.HoldDuration = 0
    local ok = pcall(function() prompt:InputHoldBegin() task.wait(0.02) prompt:InputHoldEnd() end)
    prompt.HoldDuration = saved
    return ok
end

local function auto_start_loop()
    while AutoPunchEnabled do
        pcall(function()
            local now = os.clock()
            if now - LastFireTime < FIRE_COOLDOWN then return end
            local character = LocalPlayer.Character
            if not character then return end
            local hrp = character:FindFirstChild("HumanoidRootPart")
            if not hrp then return end
            local prompt = find_punch_prompt()
            if not prompt then task.wait(5) return end
            local pp = prompt.Parent
            if pp and pp:IsA("BasePart") then
                if (pp.Position - hrp.Position).Magnitude > (prompt.MaxActivationDistance or 10) then task.wait(0.5) return end
            end
            local inGame = false
            pcall(function()
                local pg = LocalPlayer:FindFirstChild("PlayerGui")
                if pg then
                    local ui = pg:FindFirstChild("PunchUI") or pg:FindFirstChild("StrengthUI") or pg:FindFirstChild("StrengthTesterUI")
                    if ui and ui.Enabled then inGame = true end
                end
            end)
            if inGame then task.wait(0.3) return end
            LastFireTime = now
            fire_prompt(prompt)
        end)
        task.wait(0.5)
    end
    AutoStartLoop = nil
end

local function toggle_auto_punch(arg)
    local enabled = arg and true or false
    if enabled then
        local flag = false
        for _, item in ipairs({ "IsGuard", "isGuard", "Isguard" }) do
            local a = LocalPlayer:GetAttribute(item)
            if a == true or a == 1 or a == "true" then flag = true break end
        end
        if not flag then
            pcall(function() warn("[Auto Punch Machine] You need to be a guard") end)
            AutoPunchEnabled = false
            PlayToggleSound()
            return false
        end
    end
    AutoPunchEnabled = enabled
    local ok, result = pcall(function() return require(ReplicatedStorage.Modules.StrengthTester) end)
    if not ok or type(result) ~= "table" then PlayToggleSound() return false end
    if enabled then
        if not _AutoPunchOriginal then _AutoPunchOriginal = result.Start end
        result.Start = function(arg2)
            local v2 = _AutoPunchOriginal(arg2)
            if typeof(arg2) == "table" and arg2.SpinComplete then
                task.delay(0.03, function()
                    if AutoPunchEnabled and arg2.SpinComplete and arg2.SpinComplete.Parent then
                        pcall(function()
                            arg2.SpinComplete:Fire({ PowerLevel = 5, Percent = 100, QteSuccess = true })
                        end)
                    end
                end)
            end
            return v2
        end
        if AutoStartLoop then task.cancel(AutoStartLoop) end
        CachedPrompt = nil
        LastFireTime = 0
        AutoStartLoop = task.spawn(auto_start_loop)
    else
        if _AutoPunchOriginal then result.Start = _AutoPunchOriginal _AutoPunchOriginal = nil end
        if AutoStartLoop then task.cancel(AutoStartLoop) AutoStartLoop = nil end
        CachedPrompt = nil
        LastFireTime = 0
    end
    PlayToggleSound()
    return true
end

PunchBtn.MouseButton1Click:Connect(function()

    end

local AutoThrow = { Enabled = false, MobileButton = nil }

local function AutoThrow_getClosestTarget()
    local char = LocalPlayer.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local closest, shortest = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local thrp = p.Character:FindFirstChild("HumanoidRootPart")
            local thum = p.Character:FindFirstChildOfClass("Humanoid")
            if thrp and thum and thum.Health > 0 then
                local d = (thrp.Position - hrp.Position).Magnitude
                if d < shortest then shortest = d closest = thrp end
            end
        end
    end
    return closest
end

local function AutoThrow_executeThrow()
    local char = LocalPlayer.Character
    if not char then return end
    local tool = char:FindFirstChildOfClass("Tool")
    if not tool then return end
    local target = AutoThrow_getClosestTarget()
    local dir = target and (target.Position - char.HumanoidRootPart.Position).Unit or char.HumanoidRootPart.CFrame.LookVector
    pcall(function()
        local remote = tool:FindFirstChild("RemoteEvent") or ReplicatedStorage:FindFirstChild("ThrowRemote", true)
        if remote then remote:FireServer(dir, 100) else tool:Activate() end
    end)
end

-- 14. Silent Aim
local SilentAimEnabled = false
local _hooked = false
local function install_hook()
	if _hooked then return end
	_hooked = true
	pcall(function()
		local GunFunctions = require(ReplicatedStorage:WaitForChild("Modules"):WaitForChild("GunFunctions"))
		local firedGun = GunFunctions.FiredGun
		GunFunctions.FiredGun = function(arg, arg2, arg3, ...)
			if not SilentAimEnabled or arg ~= LocalPlayer.Character then
				return firedGun(arg, arg2, arg3, ...)
			end
			arg3 = arg3 or {}
			local hrp = arg and arg:FindFirstChild("HumanoidRootPart")
			if not hrp then return firedGun(arg, arg2, arg3, ...) end
			local pos = hrp.Position
			pcall(function()
				local a = arg:GetAttribute("HoldingWeapon")
				if a then
					local w = arg:FindFirstChild(a)
					if w then
						local ff = w:FindFirstChild("FireFrom")
						if ff then pos = ff.Position end
					end
				end
			end)
			local isGuard = LocalPlayer:GetAttribute("IsGuard")
			local live = Workspace:FindFirstChild("Live")
			local target = nil
			if live then
				local best = math.huge
				for _, child in pairs(live:GetChildren()) do
					if child ~= arg and not child:FindFirstChild("Dead") and not child:FindFirstChild("IFrame") then
						local hum = child:FindFirstChild("Humanoid")
						if not (hum and hum.Health <= 0) then
							local head = child:FindFirstChild("Head") or child:FindFirstChild("HumanoidRootPart") or child:FindFirstChildWhichIsA("BasePart", true)
							if head then
								local h2 = child:FindFirstChild("HumanoidRootPart")
								local ok2 = false
								if child:FindFirstChild("Enemy") or string.match(string.lower(child.Name), "peabert") then ok2 = true
								elseif isGuard then
									if child:FindFirstChild("GuardCanKill") or (h2 and h2:FindFirstChild("GuardCanKillLockOn")) then ok2 = true end
								else
									local plr = Players:GetPlayerFromCharacter(child)
									if (plr and plr ~= LocalPlayer and plr:GetAttribute("IsGuard"))
										or child.Name:match("Rebel") or child.Name:match("Guard")
										or child:FindFirstChild("PlayerCanKill") or child:FindFirstChild("TypeOfGuard") then ok2 = true end
								end
								local tog = isGuard and child:FindFirstChild("TypeOfGuard")
								if tog then tog = not (h2 and h2:FindFirstChild("GuardCanKillLockOn")) end
								if tog and not child:FindFirstChild("GuardCanKill") then ok2 = false end
								if ok2 then
									local mag = (head.Position - pos).Magnitude
									if mag < best and mag < 1000 then best = mag target = head end
								end
							end
						end
					end
				end
			end
			if target then
				arg3.CustomFireFrom = true
				arg3.spread = 0
				arg2 = target.Position
			end
			return firedGun(arg, arg2, arg3, ...)
		end
	end)
end
local function toggle_silent_aim(en)
	SilentAimEnabled = en and true or false
	if SilentAimEnabled then install_hook() end
	PlayToggleSound()
	return true
end

    SilentAimEnabled = not SilentAimEnabled
    if SilentAimEnabled then
        TweenService:Create(SilentSwitchBG, EASE_OUT, {BackgroundColor3 = CyanColor}):Play()
        TweenService:Create(SilentSwitchCircle, EASE_OUT, {Position = UDim2.new(1, -14, 0.5, -6), BackgroundColor3 = Color3.fromRGB(15, 15, 20)}):Play()

        TweenService:Create(SilentSwitchCircle, EASE_OUT, {Position = UDim2.new(0, 2, 0.5, -6), BackgroundColor3 = T.KNOB_OFF}):Play()
    end

end)

-- 15. Auto Escape
EscapeBtn.MouseButton1Click:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local target = nil
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("SpawnLocation") and v.Enabled then
            target = v.CFrame + Vector3.new(0, 3, 0)
            break
        end
    end
    if not target then
        for _, v in pairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                local n = v.Name:lower()
                if n:find("spawn") or n:find("base") or n:find("lobby") or n:find("safezone") then
                    target = v.CFrame + Vector3.new(0, 4, 0)
                    break
                end
            end
        end
    end
    if target then root.CFrame = target else root.CFrame = CFrame.new(0, 100, 0) end
end)
